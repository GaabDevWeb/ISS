-- Parte 1
CREATE TABLE departamentos (
    id_departamento integer,
    departamento text,
    cidade text,
    CONSTRAINT pk_departamentos PRIMARY KEY (id_departamento),
    CONSTRAINT unique_departamento_cidade UNIQUE (departamento, cidade)
);

CREATE TABLE funcionarios (
    matricula integer,
    primeiro_nome text,
    ultimo_nome text,
    salario numeric(10,2),
    id_departamento integer REFERENCES departamentos (id_departamento),
    CONSTRAINT pk_funcionarios PRIMARY KEY (matricula)
);

SELECT * 
FROM departamentos; 

SELECT * 
FROM funcionarios;

SELECT *
FROM funcionarios 
  JOIN departamentos
ON funcionarios.id_departamento = departamentos.id_departamento
ORDER BY funcionarios.id_departamento;

SELECT *
FROM funcionarios 
  INNER JOIN departamentos
ON funcionarios.id_departamento = departamentos.id_departamento
ORDER BY funcionarios.id_departamento;

SELECT *
FROM funcionarios,  
  departamentos
WHERE funcionarios.id_departamento = departamentos.id_departamento
ORDER BY funcionarios.id_departamento;

SELECT distrito_2020.id as  d20_id,
       distrito_2020.escola_2020 as d20_sch_2020,
       distrito_2035.escola_2035 as d35_sch_2035
FROM distrito_2020 
LEFT JOIN distrito_2035
ON distrito_2020.id = distrito_2035.id
ORDER BY distrito_2020.id;

SELECT distrito_2020.id as "District 2020 ID",
       distrito_2020.escola_2020 as "District 2020 School",
       distrito_2035.escola_2035 as "District 2035 School"
FROM distrito_2020 
LEFT JOIN distrito_2035
ON distrito_2020.id = distrito_2035.id
ORDER BY distrito_2020.id;

SELECT d20.id, 
       d20.escola_2020,
       d35.escola_2035
FROM distrito_2020 AS d20 
LEFT JOIN distrito_2035 AS d35
ON d20.id = d35.id
ORDER BY d20.id;

SELECT d20.id,
       d20.escola_2020,
       en.inscricao,
       gr.notas
FROM distrito_2020 AS d20 
   JOIN distrito_2020_inscricao AS en
      ON d20.id = en.id
   JOIN distrito_2020_notas AS gr
      ON d20.id = gr.id
ORDER BY d20.id;
